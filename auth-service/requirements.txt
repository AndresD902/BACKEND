# ── NOTAS DE INSTALACIÓN ────────────────────────────────────
# 1. Requiere Node.js >= 18 (para compatibilidad con Node16 module resolution)
# 2. Requiere PostgreSQL >= 14 corriendo y accesible
# 3. Configurar .env antes de ejecutar (ver .env.example)
# 4. Ejecutar migraciones antes del primer inicio:
#       npm run migrate
# 5. Para desarrollo:
#       npm run dev
# 6. Para producción:
#       npm run build && npm start


# ── SCRIPTS NPM ─────────────────────────────────────────────
# npm run dev              → Servidor con hot-reload (ts-node-dev)
# npm run build            → Compilar TypeScript a dist/
# npm start                → Iniciar servidor compilado
# npm test                 → Ejecutar tests con Jest
# npm run test:coverage    → Tests + reporte de cobertura
# npm run migrate          → Aplicar migraciones pendientes
# npm run migrate:down     → Revertir última migración
# npm run lint             → Verificar código con ESLint
# npm run format           → Formatear código con Prettier


# ── CONFIGURACIÓN TYPESCRIPT ────────────────────────────────
# tsconfig.json          → Compilación de producción (src/ únicamente)
# tsconfig.test.json     → Compilación de tests (src/ + tests/)
# tsconfig.migrations.json → Compilación de migraciones
#
# module:           Node16   (compatible TS 5.x → 7.x)
# moduleResolution: Node16   (resuelve exports field de paquetes)
# target:           ES2022
# strict:           true
